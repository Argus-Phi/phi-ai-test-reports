import { ToolRunningDots } from "@phi-ai/shared-ui/message-parts/tool-running-dots";
import {
  isRunningToolState,
  TOOL_PART_ABORTED_STATE,
} from "@sidecar/lib/tool-part-state";
import type {
  DynamicToolUIPart,
  UIDataTypes,
  UIMessagePart,
  UITools,
} from "ai";

type AbortedToolPart = Omit<DynamicToolUIPart, "state"> & {
  state: typeof TOOL_PART_ABORTED_STATE;
  errorText?: string;
};

type ToolPartType =
  | Extract<UIMessagePart<UIDataTypes, UITools>, { type: `tool-${string}` }>
  | DynamicToolUIPart
  | AbortedToolPart;

export interface ToolPartProps {
  part: ToolPartType;
}

// eslint-disable-next-line react-refresh/only-export-components
export function isToolPartVisible(part: ToolPartType): boolean {
  return isRunningToolState(part.state);
}

export default function ToolPart({ part }: ToolPartProps) {
  if (isRunningToolState(part.state)) {
    return <ToolRunningDots />;
  }
  return null;
}
import i18n from "i18next";

export const TOOL_PART_ABORTED_STATE = "output-aborted";

/** English fallback when i18n resolution yields nothing (pre-init). */
export const DEFAULT_ABORTED_TOOL_ERROR_TEXT = "Interrupted by user";

const ABORTED_TOOL_ERROR_KEY = "message.tool.interruptedByUser";

/**
 * Localized error text written onto a tool part when the user aborts the run.
 * `translate` is injectable purely for testing; production callers omit it and
 * get the bound `i18next.t`.
 */
export function getAbortedToolErrorText(
  translate: (key: string) => string | undefined = (key) => i18n.t(key),
): string {
  const translated = translate(ABORTED_TOOL_ERROR_KEY);
  if (
    typeof translated === "string" &&
    translated.trim().length > 0 &&
    translated !== ABORTED_TOOL_ERROR_KEY
  ) {
    return translated;
  }
  return DEFAULT_ABORTED_TOOL_ERROR_TEXT;
}

const RUNNING_TOOL_STATES = new Set([
  "input-streaming",
  "input-available",
  "approval-requested",
  "approval-responded",
]);

export function isRunningToolState(state: unknown): boolean {
  return typeof state === "string" && RUNNING_TOOL_STATES.has(state);
}

export function isAbortedToolState(state: unknown): boolean {
  return state === TOOL_PART_ABORTED_STATE;
}

export function normalizeDynamicToolPartForAbort(
  part: Record<string, unknown>,
  errorText = getAbortedToolErrorText(),
): Record<string, unknown> {
  const nextPart: Record<string, unknown> = {
    ...part,
    state: TOOL_PART_ABORTED_STATE,
    input:
      Object.prototype.hasOwnProperty.call(part, "input") &&
      part.input !== undefined
        ? part.input
        : null,
    errorText,
  };

  delete nextPart.output;
  return nextPart;
}
